## ストリーミングトーク機能

- TODO: ストリーミングトークとは何かの説明

```py
import time

from sounddevice import RawOutputStream
from voicevox_core import FRAME_RATE

STYLE_ID: voicevox_core.StyleId
synth: voicevox_core.blocking.Synthesizer
query: voicevox_core.AudioQuery

stream = await synth.streaming_synthesis(query, STYLE_ID)
_wav_header = await anext(stream)

segments = asyncio.Queue[bytes | None]()

async def render() -> None:
    try:
        async for segment in stream:
            segments.put_nowait(segment)
    finally:
        segments.put_nowait(None)

rendering = asyncio.create_task(render())

with RawOutputStream(
    samplerate=float(query.output_sampling_rate),
    channels=2 if query.output_stereo else 1,
    dtype="int16",
    latency=0.4,
) as out:
    rendering_started = time.monotonic_ns()
    while segment := await segments.get():
        _underflowed = await asyncio.to_thread(out.write, segment)
    await rendering
    estimated_remaining_playback = (
        query.frame_length() / FRAME_RATE
        - (time.monotonic_ns() - rendering_started) / 1e9
    )
    await asyncio.sleep(max(0.0, estimated_remaining_playback + 0.1))
```
